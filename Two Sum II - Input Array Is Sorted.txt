class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        int idx;
        for (int i = 0; i < nums.size(); i++) {
            auto it = lower_bound(nums.begin() + i + 1, nums.end(), target - nums[i]);
            idx = it - nums.begin();
            if (idx < nums.size() && nums[idx] == target - nums[i])
                return {i + 1, idx + 1};
        }
        return {};
    }
};
